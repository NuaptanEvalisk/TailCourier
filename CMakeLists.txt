cmake_minimum_required(VERSION 3.22)
project(TailCourier VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Widgets Network)
find_package(KF6StatusNotifierItem REQUIRED)
find_package(KF6Notifications REQUIRED)
find_package(KF6DBusAddons REQUIRED)
add_library(tailcourier-core src/storage.cpp src/backend.cpp)
target_include_directories(tailcourier-core PUBLIC src)
target_link_libraries(tailcourier-core PUBLIC Qt6::Core Qt6::Network)
target_compile_options(tailcourier-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tailcourier src/main.cpp src/window.cpp)
target_link_libraries(tailcourier PRIVATE tailcourier-core Qt6::Widgets KF6::StatusNotifierItem KF6::Notifications KF6::DBusAddons)
install(TARGETS tailcourier RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/org.kde.tailcourier.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(PROGRAMS data/tailcourier-send.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/kio/servicemenus)
install(FILES data/tailcourier.notifyrc DESTINATION ${CMAKE_INSTALL_DATADIR}/knotifications6)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 6.6 REQUIRED COMPONENTS Test DBus)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 find_package(KF6KIO REQUIRED)
 add_executable(core-test tests/core-test.cpp)
 target_link_libraries(core-test PRIVATE tailcourier-core Qt6::Test)
 add_test(NAME core COMMAND core-test)
 add_executable(gui-test tests/gui-test.cpp src/window.cpp)
 target_link_libraries(gui-test PRIVATE tailcourier-core Qt6::Widgets Qt6::Test Qt6::DBus KF6::StatusNotifierItem KF6::Notifications)
 file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-data/knotifications6)
 configure_file(data/tailcourier.notifyrc test-data/knotifications6/tailcourier.notifyrc COPYONLY)
 add_test(NAME gui COMMAND dbus-run-session -- $<TARGET_FILE:gui-test> -platform offscreen)
 set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_DATA_DIRS=${CMAKE_CURRENT_BINARY_DIR}/test-data:/usr/share" TIMEOUT 45)
 add_test(NAME single-instance COMMAND dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/single-instance.py $<TARGET_FILE:tailcourier>)
 set_tests_properties(single-instance PROPERTIES TIMEOUT 20)
 add_executable(dolphin-test tests/dolphin-test.cpp)
 target_link_libraries(dolphin-test PRIVATE Qt6::Widgets Qt6::Test KF6::KIOWidgets)
 file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-data/kio/servicemenus)
 configure_file(data/tailcourier-send.desktop test-data/kio/servicemenus/tailcourier-send.desktop COPYONLY)
 file(CHMOD ${CMAKE_CURRENT_BINARY_DIR}/test-data/kio/servicemenus/tailcourier-send.desktop
      PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
 add_test(NAME dolphin COMMAND dbus-run-session -- $<TARGET_FILE:dolphin-test> -platform offscreen)
 set_tests_properties(dolphin PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_DATA_DIRS=${CMAKE_CURRENT_BINARY_DIR}/test-data:/usr/share" TIMEOUT 20)
endif()
